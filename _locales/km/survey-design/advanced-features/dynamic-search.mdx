**ការស្វែងរកបែបឌីណាមិក** (ហៅម្យ៉ាងទៀតថា Search API) អនុញ្ញាតឱ្យវាល `select_one`, `select_multiple` ឬ `text` ផ្ទុកជម្រើសរបស់វាពីសេវាវេបពីចម្ងាយ **នៅពេលកំពុងដំណើរការ (runtime)** នៅពេលដែលអ្នកសម្ភាសន៍វាយបញ្ចូល។ នេះគឺជាវិធីសាស្ត្រដ៏ត្រឹមត្រូវនៅពេលដែលបញ្ជីជម្រើសរបស់អ្នកធំពេកមិនអាចដាក់ក្នុងឯកសារ CSV បាន ត្រូវបានធ្វើបច្ចុប្បន្នភាពញឹកញាប់ ឬមកពីមូលដ្ឋានទិន្នន័យផ្ទាល់។

---

## រូបរាង (appearance) `search-api()`

ការស្វែងរកបែបឌីណាមិកត្រូវបានកំណត់រចនាសម្ព័ន្ធតាមរយៈជួរឈរ `appearance` ដោយប្រើមុខងារ `search-api()`៖

```
search-api(method, url, post_body, value_column, display, data_path, save_path)
```

### ប៉ារ៉ាម៉ែត្រ (Parameters)

| ប៉ារ៉ាម៉ែត្រ | ការពិពណ៌នា |
|-----------|-------------|
| `method` | ត្រូវប្រើ `'POST'` ជានិច្ច |
| `url` | ចំណុចបញ្ចប់ API (endpoint) ដើម្បីសាកសួរ |
| `post_body` | តួ JSON ដែលផ្ញើទៅ API។ ប្រើ `%__input__%` ជាកន្លែងទុកសម្រាប់អត្ថបទស្វែងរកបច្ចុប្បន្នរបស់អ្នកសម្ភាសន៍ |
| `value_column` | សោ (key) នៅក្នុងវត្ថុឆ្លើយតបដើម្បីប្រើជា **តម្លៃ (value)** ដែលត្រូវរក្សាទុក |
| `display` | សោ (key) (ឬគំរូ) ដើម្បីប្រើជា **ស្លាក (label)** ដែលបង្ហាញក្នុងបញ្ជីជ្រើសរើស។ គាំទ្រកន្លែងទុក `##key##` និងកន្សោម `@{func}` |
| `data_path` | JSONPath ទៅកាន់អារេនៃវត្ថុលទ្ធផលនៅក្នុងការឆ្លើយតប (ឧទាហរណ៍៖ `$.data`, `$.hits.hits.*._source`) |
| `save_path` | ឈ្មោះដែលការឆ្លើយតបដើមត្រូវបានរក្សាទុកសម្រាប់ប្រើប្រាស់ដោយវាលផ្សេងទៀត |

## ឧទាហរណ៍មូលដ្ឋាន

ការស្វែងរកមូលដ្ឋានសុខាភិបាលដែលអ្នកសម្ភាសន៍វាយបញ្ចូលផ្នែកខ្លះនៃឈ្មោះមូលដ្ឋាន៖

| type | name | label | appearance |
|------|------|-------|------------|
| select_one | facility | ជ្រើសរើសមូលដ្ឋានសុខាភិបាល | `search-api('POST', 'https://api.example.com/facilities/search', '{"query": "%__input__%"}', 'id', 'name', '$.results', 'facility_data')` |

API ទទួល `{"query": "nair"}` នៅពេលអ្នកសម្ភាសន៍វាយ "nair" ហើយត្រឡប់មកវិញនូវ៖

```json
{
  "results": [
    {"id": "HF001", "name": "Nairobi Central Clinic"},
    {"id": "HF002", "name": "Nairobi West Hospital"}
  ]
}
```

បញ្ជីជ្រើសរើសបង្ហាញ `Nairobi Central Clinic` និង `Nairobi West Hospital` ហើយតម្លៃដែលបានរក្សាទុកគឺ `HF001` ឬ `HF002`។

## ការកំណត់ទម្រង់បង្ហាញកម្រិតខ្ពស់

### ការប្រើប្រាស់គំរូ `##key##`

បង្ហាញវាលច្រើនក្នុងស្លាក៖

```
search-api('POST', 'https://api.example.com/search', '{"q": "%__input__%"}', 'id', '##name## (##district##)', '$.data', 'res')
```

បង្ហាញជា៖ `Nairobi Central Clinic (Nairobi)`។

### ការប្រើប្រាស់កន្សោម `@{func}`

អនុវត្តតក្កវិជ្ជាមានលក្ខខណ្ឌនៅក្នុងស្លាកបង្ហាញ៖

```
search-api('POST', 'https://api.example.com/search', '{"q": "%__input__%"}', 'id',
  '@{if_else(eq("##status##", "active"), "✓ ##name##", "✗ ##name##")}',
  '$.data', 'res')
```

លទ្ធផលដែលសកម្មបង្ហាញ `✓ ឈ្មោះគ្លីនិក` ហើយលទ្ធផលដែលមិនសកម្មបង្ហាញ `✗ ឈ្មោះគ្លីនិក`។

---

## ការកំណត់តម្លៃលំនាំដើម៖ `search-default-api()`

ប្រើ `search-default-api()` បន្ទាប់ពី `search-api()` ដើម្បីបំពេញវាលជាមុនជាមួយនឹងជម្រើសលំនាំដើមដែលផ្ទុកពីការហៅ API ដាច់ដោយឡែក (ឧទាហរណ៍៖ នៅពេលកែសម្រួលកំណត់ត្រាដែលមានស្រាប់)៖

```
appearance: search-api(...) search-default-api('POST', 'https://api.example.com/get', '{"id": "##saved_id##"}', 'id', 'name', '$.item')
```

## ឧបករណ៍បំបែកផ្ទាល់ខ្លួនសម្រាប់ select_multiple៖ `search-default-separator()`

សម្រាប់វាល `select_multiple` កំណត់ពីរបៀបដែលតម្លៃដែលបានជ្រើសរើសច្រើនត្រូវបានភ្ជាប់គ្នានៅក្នុងខ្សែអក្សរដែលបានរក្សាទុក៖

```
appearance: search-api(...) search-default-separator(' || ')
```

ឧបករណ៍បំបែកលំនាំដើមគឺចន្លោះ (space)។

## ប្រភេទសំណួរដែលគាំទ្រ

| ប្រភេទសំណួរ | ករណីប្រើប្រាស់ |
|---------------|----------|
| `select_one` | ការជ្រើសរើសតែមួយពីលទ្ធផលស្វែងរក |
| `select_multiple` | ការជ្រើសរើសច្រើនពីលទ្ធផលស្វែងរក |
| `text` | បំពេញដោយស្វ័យប្រវត្តិ (Autocomplete) — អ្នកសម្ភាសន៍វាយបញ្ចូលដោយសេរី ប៉ុន្តែអាចជ្រើសរើសការណែនាំបាន |

## ការប្រើប្រាស់ទិន្នន័យឆ្លើយតបដែលបានរក្សាទុក

នៅពេលអ្នកសម្ភាសន៍ជ្រើសរើសលទ្ធផល rtSurvey រក្សាទុកវត្ថុដែលត្រូវគ្នាទាំងស្រុងពីការឆ្លើយតបក្រោមឈ្មោះ `save_path`។ វាលផ្សេងទៀតអាចអានពីវាបានតាមពីរវិធី។

### វិធីទី ១ — ការរកមើលតាម `pulldata()`

ប្រើ `pulldata('save_path', 'key')` ដើម្បីអានវាលកម្រិតកំពូលពីវត្ថុដែលបានរក្សាទុក៖

| type | name | label | calculation |
|------|------|-------|-------------|
| select_one | facility | ជ្រើសរើសមូលដ្ឋាន | |
| calculate | facility_district | | `pulldata('facility_data', 'district')` |
| calculate | facility_type | | `pulldata('facility_data', 'type')` |
| calculate | facility_beds | | `pulldata('facility_data', 'beds')` |

រូបរាង (appearance) នៅលើជួរ `select_one` ត្រូវតែរួមបញ្ចូល `search-api(..., 'facility_data')` ជាមួយនឹង `save_path` ដែលត្រូវគ្នា។

សម្រាប់ **វាល JSON ដែលមានស្រទាប់បន្តគ្នា** សូមបញ្ចូលគ្នាជាមួយ `substr-jsonpath()`៖

```
substr-jsonpath(pulldata('facility_data', 'location'), '$.district')
```

## ការស្វែងរកឯកសារក្នុងស្រុកជាមួយ search()

`search()` អនុវត្តការរកមើលភាពស្មើគ្នានៃជួរឈរក្រៅបណ្តាញ ប្រឆាំងនឹងឯកសារ SQLite `.db` ដែលបានភ្ជាប់មកជាមួយ — មិនត្រូវការការភ្ជាប់បណ្តាញ និងមិនត្រូវការសំណួរ SQL ឡើយ។ ប្រើវា عندما ទិន្នន័យយោងរបស់អ្នកត្រូវបានខ្ចប់ជាមួយទម្រង់បែបបទ ហើយអ្នកត្រូវការការរកមើលសាមញ្ញ "ស្វែងរកជួរដេកទាំងអស់ដែលជួរឈរស្មើនឹងតម្លៃ"។

### ហត្ថលេខា (Signature)

```
search(path, 'matches', 'list_name', ${field})
```

### ប៉ារ៉ាម៉ែត្រ (Parameters)

| ប៉ារ៉ាម៉ែត្រ | ការពិពណ៌នា |
|-----------|-------------|
| `path` | ផ្លូវទៅកាន់តារាង SQLite៖ `concat(${family_path}, '/file.db::table')` |
| `'matches'` | ប្រតិបត្តិករផ្គូផ្គង — ត្រឡប់ជួរដេកដែលជួរឈរស្មើនឹងតម្លៃ |
| `'list_name'` | ឈ្មោះជួរឈរដែលត្រូវផ្គូផ្គង |
| `${field}` | តម្លៃដែលត្រូវផ្គូផ្គង ជាទូទៅគឺជាការយោងវាល |

### ឧទាហរណ៍

បំពេញ `select_one` ពីមូលដ្ឋានទិន្នន័យក្នុងស្រុកដោយផ្គូផ្គងតម្លៃនៃ `${d307}` ជាមួយជួរឈរ `list_name` នៅក្នុងតារាង `externalData` នៃ `d307.db`៖

| type | name | label | appearance |
|------|------|-------|------------|
| select_one | facility | ជ្រើសរើសមូលដ្ឋាន | `search(concat(${family_path}, '/d307.db::externalData'), 'matches', 'list_name', ${d307})` |

នៅពេលដែល `${d307}` ស្មើនឹង `"HF001"`, `search()` ត្រឡប់គ្រប់ជួរដេកក្នុង `externalData` ដែល `list_name = 'HF001'` — ទាំងស្រុងពីឯកសារដែលបានភ្ជាប់មកជាមួយ ដោយមិនមានការហៅតាមបណ្តាញឡើយ។

### search() ប្រៀបធៀបនឹង rawquery autocomplete

| លក្ខណៈពិសេស | `search()` | `rawquery` autocomplete |
|---------|-----------|------------------------|
| សំណួរ SQL | ទេ — ផ្គូផ្គងភាពស្មើគ្នានៃជួរឈរប៉ុណ្ណោះ | បាទ — `SELECT … FROM …` ពេញលេញ |
| ការបញ្ចូលប៉ារ៉ាម៉ែត្រ | ផ្គូផ្គងតម្លៃជួរឈរ | បញ្ចូលក្នុងខ្សែអក្សរ SQL |
| ការបញ្ចូលអត្ថបទសេរី | បាទ — អ្នកសម្ភាសន៍អាចវាយបញ្ចូលដោយសេរី | ទេ — ត្រូវតែជ្រើសរើសពីបញ្ជី |
| UNION / ច្រើនតារាង | មិនគាំទ្រ | គាំទ្រ |

### ពេលណាត្រូវប្រើនីមួយៗ

- ប្រើ **`search()`** នៅពេលអ្នកត្រូវការការផ្គូផ្គងភាពស្មើគ្នាដោយផ្ទាល់ប្រឆាំងនឹងឯកសារ `.db` ដែលបានភ្ជាប់មកជាមួយ ទម្រង់បែបបទត្រូវតែដំណើរការក្រៅបណ្តាញ ហើយអ្នកចង់អនុញ្ញាតឱ្យបញ្ចូលអត្ថបទសេរី។
- ប្រើ **`rawquery` autocomplete** នៅពេលអ្នកត្រូវការភាពបត់បែននៃ SQL (JOINs, UNIONs, ជួរឈរគណនា) ឬត្រូវសាកសួរឆ្លងតារាងច្រើន ហើយទម្រង់បែបបទមានការភ្ជាប់បណ្តាញដែលអាចទុកចិត្តបាន។

## ការអនុវត្តល្អបំផុត (Best Practices)

១. ធានាថាចំណុចបញ្ចប់ API របស់អ្នកឆ្លើយតបក្នុងរយៈពេល ១-២ វិនាទី — API យឺតធ្វើឱ្យការស្វែងរកហាក់ដូចជាមិនឆ្លើយតប។
២. ប្រើ `%__input__%` នៅក្នុង `post_body` ដើម្បីឱ្យ API ត្រឡប់តែលទ្ធផលដែលត្រូវគ្នា មិនមែនសំណុំទិន្នន័យទាំងមូលឡើយ។
៣. បង្កើតសន្ទស្សន៍ (Index) លើវាលស្វែងរកនៅផ្នែកម៉ាស៊ីនមេ (ឧទាហរណ៍៖ Elasticsearch, សន្ទស្សន៍អត្ថបទពេញលេញនៃមូលដ្ឋានទិន្នន័យ) សម្រាប់ការឆ្លើយតបរហ័ស។
៤. កំណត់លទ្ធផលត្រឹម ២០–៥០ ធាតុក្នុងមួយសំណួរ — ការត្រឡប់លទ្ធផលរាប់ពាន់បំផ្លាញគោលបំណងនៃការស្វែងរក។
៥. រួមបញ្ចូលតម្រូវការប្រវែងបញ្ចូលអប្បបរមានៅក្នុង API ដើម្បីជៀសវាងការបង្កឱ្យមានសំណួរទូលំទូលាយលើការបញ្ចូលតួអក្សរតែមួយ។

## ដែនកំណត់

- ការស្វែងរកបែបឌីណាមិកតម្រូវឱ្យមានការភ្ជាប់បណ្តាញ — វាមិនដំណើរការក្រៅបណ្តាញឡើយ។
- កន្លែងទុក `%__input__%` ត្រូវបានបញ្ចូលដូចដើម សូមសម្អាតការបញ្ចូលនៅផ្នែកម៉ាស៊ីនមេ ដើម្បីការពារការវាយប្រហារបែប injection។
- កន្សោមបង្ហាញ `@{func}` ស្មុគស្មាញអាចមានការគាំទ្រមានកម្រិតនៅគ្រប់កំណែ rtSurvey client ទាំងអស់។
